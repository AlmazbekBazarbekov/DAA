public class Solution extends VersionControl {
    public int firstBadVersion(int n) {
        int leftBorder = 1;
        int rightBorder = n;
        while(leftBorder < rightBorder){
            int mid = leftBorder + (rightBorder - leftBorder) / 2;
            if(isBadVersion(mid)){
                rightBorder = mid;
            }
            else {
                leftBorder = mid + 1;
            }
        }return leftBorder;
    }
}
