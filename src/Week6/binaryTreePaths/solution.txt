class Solution {
    public List<String> binaryTreePaths(TreeNode root) {
        List<String> result = new ArrayList<>();
        if(root == null) return result;
        check(root, String.valueOf(root.val), result);

        return result;
    }

    public void check(TreeNode root, String path, List<String> result) {
        if(root.right == null && root.left == null) {
            result.add(path);
            return;
        };
        if(root.right != null) {
            check(root.right, path + "->" + root.right.val, result);
        }

        if(root.left != null) {
            check(root.left, path + "->"+ root.left.val, result);
        }
    }
}
